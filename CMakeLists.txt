cmake_minimum_required(VERSION 3.21)
project(SLINGSHIFT VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(SLINGSHIFT_BUILD_GAME "Build the raylib game" ON)
option(SLINGSHIFT_BUILD_TESTS "Build headless physics tests" ON)
option(SLINGSHIFT_SANITIZE "Enable address and undefined-behaviour sanitizers" OFF)
option(SLINGSHIFT_TEST_HOOKS "Expose opt-in browser QA hooks (never for release)" OFF)

add_library(sling_physics src/physics.cpp src/levels.cpp)
target_include_directories(sling_physics PUBLIC src)
if(MSVC)
    target_compile_options(sling_physics PRIVATE /W4 /permissive-)
else()
    target_compile_options(sling_physics PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SLINGSHIFT_SANITIZE AND NOT MSVC AND NOT EMSCRIPTEN)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    target_compile_options(sling_physics PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

if(SLINGSHIFT_BUILD_TESTS)
    enable_testing()
    add_executable(physics_tests tests/physics_tests.cpp)
    target_link_libraries(physics_tests PRIVATE sling_physics)
    add_test(NAME physics_and_all_levels COMMAND physics_tests)
    add_executable(save_tests tests/save_tests.cpp src/save.cpp)
    target_include_directories(save_tests PRIVATE src)
    add_test(NAME progress_and_settings COMMAND save_tests)
    if(MINGW)
        target_link_options(physics_tests PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_options(save_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_executable(solve_levels EXCLUDE_FROM_ALL tools/solve_levels.cpp)
    target_link_libraries(solve_levels PRIVATE sling_physics)
endif()

if(SLINGSHIFT_BUILD_GAME)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    if(EMSCRIPTEN)
        set(PLATFORM Web CACHE STRING "" FORCE)
        set(GRAPHICS GRAPHICS_API_OPENGL_ES2 CACHE STRING "" FORCE)
    endif()
    add_subdirectory(vendor/raylib EXCLUDE_FROM_ALL)
    if(EMSCRIPTEN)
        target_compile_options(raylib PRIVATE "-include${CMAKE_CURRENT_SOURCE_DIR}/web/legacy_glfw.h")
    endif()
    add_executable(slingshift src/main.cpp src/game.cpp src/art.cpp src/audio.cpp src/save.cpp)
    target_link_libraries(slingshift PRIVATE sling_physics raylib)
    target_compile_definitions(slingshift PRIVATE $<$<BOOL:${SLINGSHIFT_TEST_HOOKS}>:SLINGSHIFT_TEST_HOOKS>)
    if(WIN32)
        target_sources(slingshift PRIVATE assets/windows.rc)
        target_include_directories(slingshift PRIVATE assets)
        set_target_properties(slingshift PROPERTIES WIN32_EXECUTABLE TRUE)
        if(MSVC)
            target_link_options(slingshift PRIVATE /ENTRY:mainCRTStartup)
        endif()
    endif()
    if(MSVC)
        target_compile_options(slingshift PRIVATE /W4 /permissive-)
    else()
        target_compile_options(slingshift PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(MINGW)
        target_link_options(slingshift PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    if(NOT EMSCRIPTEN)
        add_executable(slingshift_artwork EXCLUDE_FROM_ALL tools/make_artwork.cpp src/art.cpp)
        target_include_directories(slingshift_artwork PRIVATE src)
        target_link_libraries(slingshift_artwork PRIVATE raylib sling_physics)
    endif()
    if(EMSCRIPTEN)
        target_sources(slingshift PRIVATE src/web_input.cpp)
        set_target_properties(slingshift PROPERTIES OUTPUT_NAME index SUFFIX ".html")
        file(GLOB_RECURSE GAME_ASSETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/*")
        set_property(TARGET slingshift APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html" ${GAME_ASSETS})
        target_link_options(slingshift PRIVATE
            "--shell-file=${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html"
            "--preload-file=${CMAKE_CURRENT_SOURCE_DIR}/assets@assets"
            -sMINIFY_HTML=0
            -sUSE_GLFW=3 -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=33554432
            -sMAXIMUM_MEMORY=268435456 -sASSERTIONS=0 -sENVIRONMENT=web
            -sFILESYSTEM=1 -sEXIT_RUNTIME=0)
        add_custom_command(TARGET slingshift POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon.png $<TARGET_FILE_DIR:slingshift>/icon.png)
    else()
        add_custom_command(TARGET slingshift POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory
                ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:slingshift>/assets)
    endif()
    install(TARGETS slingshift RUNTIME DESTINATION .)
    if(NOT EMSCRIPTEN)
        install(DIRECTORY assets DESTINATION .)
    endif()
endif()
